# dumpdata
python -Xutf8 manage.py dumpdata --format=json -o mydata.json
python -Xutf8 manage.py dumpdata --format=xml -o mydata.xml
or
python -Xutf8 manage.py dumpdata --indent 2 --format=json -o mydata.json
python -Xutf8 manage.py dumpdata --indent 2 --format=xml -o mydata.xml
better
python -Xutf8 manage.py dumpdata --indent 2 --exclude contenttypes --format=json -o mydata.json

python -Xutf8 manage.py dumpdata app --format=json -o mydata.json

## Команда для создания файла дампа с данными из таблицы по модели Post приложения app:
python -Xutf8 manage.py dumpdata app.post --format=json -o mydata.json
